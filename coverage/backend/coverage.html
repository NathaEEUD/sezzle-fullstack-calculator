// Command calculator runs the calculation service. It only reads env, wires
// the handler and shuts down; the logic lives in internal/.
package main

import (
        "context"
        "errors"
        "log/slog"
        "net/http"
        "os"
        "os/signal"
        "syscall"
        "time"

        "github.com/NathaEEUD/sezzle-fullstack-calculator/backend/internal/httpapi"
)

func main() {
        logger := slog.New(slog.NewJSONHandler(os.Stdout, nil))
        if err := run(logger); err != nil {
                logger.Error("server stopped", "error", err)
                os.Exit(1)
        }
}

func run(logger *slog.Logger) error {
        port := os.Getenv("PORT")
        if port == "" {
                port = "8080"
        }
        staticDir := os.Getenv("STATIC_DIR")

        srv := &http.Server{
                Addr:              ":" + port,
                Handler:           httpapi.NewHandler(logger, httpapi.WithStaticDir(staticDir)),
                ReadHeaderTimeout: 5 * time.Second,
                ReadTimeout:       10 * time.Second,
                WriteTimeout:      10 * time.Second,
                IdleTimeout:       60 * time.Second,
        }

        ctx, stop := signal.NotifyContext(context.Background(), syscall.SIGINT, syscall.SIGTERM)
        defer stop()

        errs := make(chan error, 1)
        go func() {
                logger.Info("listening", "addr", srv.Addr, "static_dir", staticDir)
                errs <- srv.ListenAndServe()
        }()

        select {
        case err := <-errs:
                return err
        case <-ctx.Done():
        }

        logger.Info("shutting down")
        shutdownCtx, cancel := context.WithTimeout(context.Background(), 10*time.Second)
        defer cancel()
        if err := srv.Shutdown(shutdownCtx); err != nil {
                return err
        }
        if err := <-errs; !errors.Is(err, http.ErrServerClosed) {
                return err
        }
        return nil
}

// Package calc evaluates calculations: one operation applied to its operands.
// It is pure maths and knows nothing about HTTP.
package calc

import "fmt"

type operation struct {
        arity int
        apply func(operands []float64) float64
}

// registry holds every operation the service can evaluate. Adding an
// operation is one entry here and one table-test row.
var registry = map[string]operation{
        "add": {arity: 2, apply: func(o []float64) float64 { return o[0] + o[1] }},
}

// Evaluate applies the named operation to its operands and returns the raw
// result, or a *Error.
func Evaluate(name string, operands []float64) (float64, error) {
        operation, ok := registry[name]
        if !ok {
                return 0, &Error{Code: CodeUnknownOperation, Message: fmt.Sprintf("unknown operation %q", name)}
        }
        if len(operands) != operation.arity {
                return 0, &Error{
                        Code:    CodeInvalidOperandCount,
                        Message: fmt.Sprintf("expected %d operands for %s, got %d", operation.arity, name, len(operands)),
                }
        }
        return operation.apply(operands), nil
}

package calc

// Code is the stable name of a calc failure. Callers map it to a response;
// calc itself knows nothing about transport.
type Code string

const (
        CodeUnknownOperation    Code = "UNKNOWN_OPERATION"
        CodeInvalidOperandCount Code = "INVALID_OPERAND_COUNT"
)

// Error is every failure Evaluate returns. Message names the specific cause.
type Error struct {
        Code    Code
        Message string
}

func (e *Error) Error() string { return e.Message }

// Is makes errors.Is match any *Error with the same Code, so the sentinels
// below work whatever the message.
func (e *Error) Is(target error) bool {
        t, ok := target.(*Error)
        return ok && t.Code == e.Code
}

// Sentinels for errors.Is.
var (
        ErrUnknownOperation    = &Error{Code: CodeUnknownOperation}
        ErrInvalidOperandCount = &Error{Code: CodeInvalidOperandCount}
)

package httpapi

import (
        "encoding/json"
        "net/http"

        "github.com/NathaEEUD/sezzle-fullstack-calculator/backend/internal/calc"
)

type calculationRequest struct {
        Operation string    `json:"operation"`
        Operands  []float64 `json:"operands"`
}

type calculationResponse struct {
        Operation string    `json:"operation"`
        Operands  []float64 `json:"operands"`
        Result    float64   `json:"result"`
}

func (a *api) calculate(w http.ResponseWriter, r *http.Request) {
        var req calculationRequest
        if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
                a.writeError(w, r, &requestError{code: codeInvalidJSON, message: "request body is not valid JSON"})
                return
        }

        result, err := calc.Evaluate(req.Operation, req.Operands)
        if err != nil {
                a.writeError(w, r, err)
                return
        }
        a.writeJSON(w, r, http.StatusOK, calculationResponse{Operation: req.Operation, Operands: req.Operands, Result: result})
}

package httpapi

import (
        "errors"
        "net/http"

        "github.com/NathaEEUD/sezzle-fullstack-calculator/backend/internal/calc"
)

const (
        codeInvalidJSON   = "INVALID_JSON"
        codeInternalError = "INTERNAL_ERROR"
)

// requestError is a bad request found before any calculation runs.
type requestError struct {
        code    string
        message string
}

func (e *requestError) Error() string { return e.message }

type envelope struct {
        Error envelopeError `json:"error"`
}

type envelopeError struct {
        Code    string `json:"code"`
        Message string `json:"message"`
}

// statusFor is the one mapping from an error to its status and envelope.
func statusFor(err error) (int, envelopeError) {
        var reqErr *requestError
        if errors.As(err, &reqErr) {
                return http.StatusBadRequest, envelopeError{Code: reqErr.code, Message: reqErr.message}
        }
        var calcErr *calc.Error
        if errors.As(err, &calcErr) {
                switch calcErr.Code {
                case calc.CodeUnknownOperation, calc.CodeInvalidOperandCount:
                        return http.StatusBadRequest, envelopeError{Code: string(calcErr.Code), Message: calcErr.Message}
                }
        }
        return http.StatusInternalServerError, envelopeError{Code: codeInternalError, Message: "internal error"}
}

// writeError answers with the envelope for err. Go error text goes to the log,
// never to the response.
func (a *api) writeError(w http.ResponseWriter, r *http.Request, err error) {
        status, body := statusFor(err)
        if status >= http.StatusInternalServerError {
                a.logger.Error("request failed", "method", r.Method, "path", r.URL.Path, "error", err)
        }
        a.writeJSON(w, r, status, envelope{Error: body})
}

// Package httpapi exposes the calculator over HTTP: routing, decoding and the
// error envelope. HTTP status codes appear only in this package.
package httpapi

import (
        "encoding/json"
        "fmt"
        "log/slog"
        "net/http"
)

type config struct {
        staticDir string
}

// Option configures NewHandler.
type Option func(*config)

// WithStaticDir serves the built app from dir for every non-API path.
// An empty dir serves nothing, so callers can pass STATIC_DIR as it is.
func WithStaticDir(dir string) Option {
        return func(c *config) { c.staticDir = dir }
}

// NewHandler builds every route and middleware of the service.
func NewHandler(logger *slog.Logger, opts ...Option) http.Handler {
        var cfg config
        for _, opt := range opts {
                opt(&cfg)
        }

        api := &api{logger: logger}
        mux := http.NewServeMux()
        mux.HandleFunc("POST /api/v1/calculate", api.calculate)
        mux.HandleFunc("GET /healthz", api.healthz)
        if cfg.staticDir != "" {
                mux.Handle("GET /", staticHandler(cfg.staticDir))
        }
        return mux
}

type api struct {
        logger *slog.Logger
}

func (a *api) healthz(w http.ResponseWriter, r *http.Request) {
        a.writeJSON(w, r, http.StatusOK, map[string]string{"status": "ok"})
}

// writeJSON encodes body before writing any header, so a body that can't be
// encoded becomes a 500 envelope instead of an empty 200.
func (a *api) writeJSON(w http.ResponseWriter, r *http.Request, status int, body any) {
        data, err := json.Marshal(body)
        if err != nil {
                a.writeError(w, r, fmt.Errorf("encode response: %w", err))
                return
        }
        w.Header().Set("Content-Type", "application/json")
        w.WriteHeader(status)
        _, _ = w.Write(append(data, '\n'))
}

package httpapi

import (
        "io/fs"
        "net/http"
        "os"
        "path"
        "strings"
)

// staticHandler serves the built app from dir. It never answers for /api/...,
// and any path that isn't a file falls back to index.html so the app can route.
func staticHandler(dir string) http.Handler {
        fsys := os.DirFS(dir)
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                if r.URL.Path == "/api" || strings.HasPrefix(r.URL.Path, "/api/") {
                        http.NotFound(w, r)
                        return
                }
                name := strings.TrimPrefix(path.Clean(r.URL.Path), "/")
                if info, err := fs.Stat(fsys, name); err != nil || info.IsDir() {
                        name = "index.html"
                }
                http.ServeFileFS(w, r, fsys, name)
        })
}
