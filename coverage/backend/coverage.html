// Command calculator runs the calculation service. It only reads env, wires
// the handler and shuts down; the logic lives in internal/.
package main

import (
        "context"
        "errors"
        "log/slog"
        "net/http"
        "os"
        "os/signal"
        "syscall"
        "time"

        "github.com/NathaEEUD/sezzle-fullstack-calculator/backend/internal/httpapi"
)

func main() {
        var level slog.Level // info when LOG_LEVEL is unset
        if v := os.Getenv("LOG_LEVEL"); v != "" {
                if err := level.UnmarshalText([]byte(v)); err != nil {
                        slog.New(slog.NewJSONHandler(os.Stdout, nil)).Error("invalid LOG_LEVEL", "value", v, "error", err)
                        os.Exit(1)
                }
        }
        logger := slog.New(slog.NewJSONHandler(os.Stdout, &slog.HandlerOptions{Level: level}))
        if err := run(logger); err != nil {
                logger.Error("server stopped", "error", err)
                os.Exit(1)
        }
}

func run(logger *slog.Logger) error {
        port := os.Getenv("PORT")
        if port == "" {
                port = "8080"
        }
        staticDir := os.Getenv("STATIC_DIR")
        allowedOrigins := os.Getenv("CORS_ALLOWED_ORIGINS")

        srv := &http.Server{
                Addr: ":" + port,
                Handler: httpapi.NewHandler(logger,
                        httpapi.WithStaticDir(staticDir),
                        httpapi.WithCORSAllowedOrigins(allowedOrigins),
                ),
                ReadHeaderTimeout: 5 * time.Second,
                ReadTimeout:       10 * time.Second,
                WriteTimeout:      10 * time.Second,
                IdleTimeout:       60 * time.Second,
        }

        ctx, stop := signal.NotifyContext(context.Background(), syscall.SIGINT, syscall.SIGTERM)
        defer stop()

        errs := make(chan error, 1)
        go func() {
                logger.Info("listening", "addr", srv.Addr, "static_dir", staticDir, "cors_allowed_origins", allowedOrigins)
                errs <- srv.ListenAndServe()
        }()

        select {
        case err := <-errs:
                return err
        case <-ctx.Done():
        }

        logger.Info("shutting down")
        shutdownCtx, cancel := context.WithTimeout(context.Background(), 10*time.Second)
        defer cancel()
        if err := srv.Shutdown(shutdownCtx); err != nil {
                return err
        }
        if err := <-errs; !errors.Is(err, http.ErrServerClosed) {
                return err
        }
        return nil
}

// Package calc evaluates calculations: one operation applied to its operands.
// It is pure maths and knows nothing about HTTP.
package calc

import (
        "fmt"
        "maps"
        "slices"
)

// Operation names an operation the service can evaluate, with its arity.
type Operation struct {
        Name  string
        Arity int
}

type entry struct {
        arity int
        apply func(operands []float64) float64
}

// registry holds every operation the service can evaluate. Adding an
// operation is one entry here and one table-test row.
var registry = map[string]entry{
        "add":      {arity: 2, apply: func(o []float64) float64 { return o[0] + o[1] }},
        "subtract": {arity: 2, apply: func(o []float64) float64 { return o[0] - o[1] }},
        "multiply": {arity: 2, apply: func(o []float64) float64 { return o[0] * o[1] }},
        "divide":   {arity: 2, apply: func(o []float64) float64 { return o[0] / o[1] }},
}

// Evaluate applies the named operation to its operands and returns the raw
// result, or a *Error.
func Evaluate(name string, operands []float64) (float64, error) {
        if name == "" {
                return 0, &Error{Code: CodeUnknownOperation, Message: "operation is required"}
        }
        found, ok := registry[name]
        if !ok {
                return 0, &Error{Code: CodeUnknownOperation, Message: fmt.Sprintf("unknown operation %q", name)}
        }
        if len(operands) != found.arity {
                return 0, &Error{
                        Code:    CodeInvalidOperandCount,
                        Message: fmt.Sprintf("expected %d operands for %s, got %d", found.arity, name, len(operands)),
                }
        }
        return found.apply(operands), nil
}

// Operations lists every operation in the registry, sorted by name.
func Operations() []Operation {
        names := slices.Sorted(maps.Keys(registry))
        operations := make([]Operation, len(names))
        for i, name := range names {
                operations[i] = Operation{Name: name, Arity: registry[name].arity}
        }
        return operations
}

package calc

// Code is the stable name of a calc failure. Callers map it to a response;
// calc itself knows nothing about transport.
type Code string

const (
        CodeUnknownOperation    Code = "UNKNOWN_OPERATION"
        CodeInvalidOperandCount Code = "INVALID_OPERAND_COUNT"
)

// Error is every failure Evaluate returns. Message names the specific cause.
type Error struct {
        Code    Code
        Message string
}

func (e *Error) Error() string { return e.Message }

// Is makes errors.Is match any *Error with the same Code, so the sentinels
// below work whatever the message.
func (e *Error) Is(target error) bool {
        t, ok := target.(*Error)
        return ok && t.Code == e.Code
}

// Sentinels for errors.Is.
var (
        ErrUnknownOperation    = &Error{Code: CodeUnknownOperation}
        ErrInvalidOperandCount = &Error{Code: CodeInvalidOperandCount}
)

import (

        "errors"

        "io"
        "net/http"

        "strings"

// maxBodyBytes caps a calculation request body.
const maxBodyBytes = 1024

// calculationRequest decodes operands as pointers so a null operand is
// rejected instead of becoming 0.
type calculationRequest struct {
        Operation string     `json:"operation"`
        Operands  []*float64 `json:"operands"`
}

type calculationResponse struct {
        Operation string    `json:"operation"`
        Operands  []float64 `json:"operands"`
        Result    float64   `json:"result"`
}

func (a *api) calculate(w http.ResponseWriter, r *http.Request) {
        operation, operands, err := decodeCalculation(w, r)
        if err != nil {
                a.writeError(w, r, err)
                return
        }

        result, err := calc.Evaluate(operation, operands)
        if err != nil {
                a.writeError(w, r, err)
                return
        }
        a.writeJSON(w, r, http.StatusOK, calculationResponse{Operation: operation, Operands: operands, Result: result})
}

// decodeCalculation reads exactly one JSON object of at most maxBodyBytes,
// with no unknown fields and no null operands.
func decodeCalculation(w http.ResponseWriter, r *http.Request) (string, []float64, error) {
        dec := json.NewDecoder(http.MaxBytesReader(w, r.Body, maxBodyBytes))
        dec.DisallowUnknownFields()

        var req calculationRequest
        if err := dec.Decode(&req); err != nil {
                return "", nil, decodeError(err)
        }
        if err := dec.Decode(&json.RawMessage{}); !errors.Is(err, io.EOF) {
                var tooLarge *http.MaxBytesError
                if errors.As(err, &tooLarge) {
                        return "", nil, decodeError(err)
                }
                return "", nil, invalidJSON("request body has data after the JSON object")
        }

        operands := make([]float64, len(req.Operands))
        for i, operand := range req.Operands {
                if operand == nil {
                        return "", nil, invalidJSON("operands must be numbers, got null")
                }
                operands[i] = *operand
        }
        return req.Operation, operands, nil
}

// decodeError names the specific cause of a decoding failure. Its message is
// written here, never copied from Go's error text.
func decodeError(err error) error {
        var tooLarge *http.MaxBytesError
        var wrongType *json.UnmarshalTypeError
        switch {
        case errors.As(err, &tooLarge):
                return invalidJSON(fmt.Sprintf("request body is over %d bytes", maxBodyBytes))
        case errors.Is(err, io.EOF):
                return invalidJSON("request body is empty")
        case errors.As(err, &wrongType):
                return typeError(wrongType)
        case strings.HasPrefix(err.Error(), "json: unknown field "):
                name := strings.TrimPrefix(err.Error(), "json: unknown field ")
                return invalidJSON("unknown field " + name)
        default:
                return invalidJSON("request body is not valid JSON")
        }
}

func typeError(err *json.UnmarshalTypeError) error {
        if literal, ok := strings.CutPrefix(err.Value, "number "); ok {
                return invalidJSON(fmt.Sprintf("operand %s is out of range", literal))
        }
        switch {
        case err.Field == "operands" && err.Type.Kind() == reflect.Slice:
                return invalidJSON("operands must be an array of numbers, got " + err.Value)
        case strings.HasPrefix(err.Field, "operands"):
                return invalidJSON("operands must be numbers, got " + err.Value)
        case err.Field == "operation":
                return invalidJSON("operation must be a string, got " + err.Value)
        default:
                return invalidJSON("request body must be a JSON object")
        }
}

import (
        "net/http"
        "strings"
)

// parseOrigins splits a comma-separated CORS_ALLOWED_ORIGINS value into its
// origins, ignoring spaces and empty entries.
func parseOrigins(list string) map[string]bool {
        origins := map[string]bool{}
        for origin := range strings.SplitSeq(list, ",") {
                if origin = strings.TrimSpace(origin); origin != "" {
                        origins[origin] = true
                }
        }
        return origins
}

// cors lets the allowed origins call the service from another origin. An
// allowed preflight is answered here with 204; a disallowed one gets a bare
// 403. Every other request reaches next, with CORS headers only for an
// allowed origin. With no allowed origins it returns next unchanged.
func cors(allowed map[string]bool, next http.Handler) http.Handler {
        if len(allowed) == 0 {
                return next
        }
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                origin := r.Header.Get("Origin")
                isPreflight := r.Method == http.MethodOptions && origin != "" &&
                        r.Header.Get("Access-Control-Request-Method") != ""
                if !allowed[origin] {
                        if isPreflight {
                                w.WriteHeader(http.StatusForbidden)
                                return
                        }
                        next.ServeHTTP(w, r)
                        return
                }

                h := w.Header()
                h.Set("Access-Control-Allow-Origin", origin)
                h.Add("Vary", "Origin")
                if isPreflight {
                        h.Set("Access-Control-Allow-Methods", "GET, POST")
                        h.Set("Access-Control-Allow-Headers", "Content-Type")
                        h.Set("Access-Control-Max-Age", "600")
                        w.WriteHeader(http.StatusNoContent)
                        return
                }
                next.ServeHTTP(w, r)
        })
}

import (

        "fmt"
        "net/http"

        "github.com/NathaEEUD/sezzle-fullstack-calculator/backend/internal/calc"

const (
        codeInvalidJSON      = "INVALID_JSON"
        codeNotFound         = "NOT_FOUND"
        codeMethodNotAllowed = "METHOD_NOT_ALLOWED"
        codeInternalError    = "INTERNAL_ERROR"
)

// requestError is a request the service refuses before any calculation
// runs: a bad request body, an unknown path or a wrong method.
type requestError struct {
        code    string
        message string
}

func (e *requestError) Error() string { return e.message }

func invalidJSON(message string) error {
        return &requestError{code: codeInvalidJSON, message: message}
}

func notFound(r *http.Request) error {
        return &requestError{
                code:    codeNotFound,
                message: fmt.Sprintf("no route for %s %s", r.Method, r.URL.Path),
        }
}

func methodNotAllowed(r *http.Request, allow string) error {
        return &requestError{
                code:    codeMethodNotAllowed,
                message: fmt.Sprintf("method %s is not allowed on %s; allowed: %s", r.Method, r.URL.Path, allow),
        }
}

type envelope struct {
        Error envelopeError `json:"error"`
}

type envelopeError struct {
        Code    string `json:"code"`
        Message string `json:"message"`
}

// requestStatus is the status of each requestError code.
var requestStatus = map[string]int{
        codeInvalidJSON:      http.StatusBadRequest,
        codeNotFound:         http.StatusNotFound,
        codeMethodNotAllowed: http.StatusMethodNotAllowed,
}

// statusFor is the one mapping from an error to its status and envelope.
func statusFor(err error) (int, envelopeError) {
        var reqErr *requestError
        if errors.As(err, &reqErr) {
                return requestStatus[reqErr.code], envelopeError{Code: reqErr.code, Message: reqErr.message}
        }
        var calcErr *calc.Error
        if errors.As(err, &calcErr) {
                switch calcErr.Code {
                case calc.CodeUnknownOperation, calc.CodeInvalidOperandCount:
                        return http.StatusBadRequest, envelopeError{Code: string(calcErr.Code), Message: calcErr.Message}
                }
        }
        return http.StatusInternalServerError, envelopeError{Code: codeInternalError, Message: "internal error"}
}

// writeError answers with the envelope for err. Go error text goes to the log,
// never to the response.
func (a *api) writeError(w http.ResponseWriter, r *http.Request, err error) {
        status, body := statusFor(err)
        if status >= http.StatusInternalServerError {
                a.logger.Error("request failed", "method", r.Method, "path", r.URL.Path, "error", err)
        }
        a.writeJSON(w, r, status, envelope{Error: body})
}

// Package httpapi exposes the calculator over HTTP: routing, decoding and the
// error envelope. HTTP status codes appear only in this package.
package httpapi

import (
        "encoding/json"
        "fmt"
        "log/slog"
        "net/http"
)

type config struct {
        staticDir      string
        allowedOrigins map[string]bool
}

// Option configures NewHandler.
type Option func(*config)

// WithStaticDir serves the built app from dir for every non-API path.
// An empty dir serves nothing, so callers can pass STATIC_DIR as it is.
func WithStaticDir(dir string) Option {
        return func(c *config) { c.staticDir = dir }
}

// WithCORSAllowedOrigins lets the comma-separated origins in list call the
// service from another origin. An empty list keeps CORS off, so callers can
// pass CORS_ALLOWED_ORIGINS as it is.
func WithCORSAllowedOrigins(list string) Option {
        return func(c *config) { c.allowedOrigins = parseOrigins(list) }
}

// NewHandler builds every route and middleware of the service.
func NewHandler(logger *slog.Logger, opts ...Option) http.Handler {
        var cfg config
        for _, opt := range opts {
                opt(&cfg)
        }

        api := &api{logger: logger}
        service := http.NewServeMux()
        service.HandleFunc("POST /api/v1/calculate", api.calculate)
        service.HandleFunc("GET /api/v1/operations", api.operations)
        service.HandleFunc("GET /healthz", api.healthz)
        var routes http.Handler = api.route(service)
        if cfg.staticDir != "" {
                app := http.NewServeMux()
                app.Handle("GET /", staticHandler(cfg.staticDir))
                routes = serviceOrApp(routes, api.route(app))
        }
        return api.logRequests(api.recoverPanics(cors(cfg.allowedOrigins, routes)))
}

type api struct {
        logger *slog.Logger
}

func (a *api) healthz(w http.ResponseWriter, r *http.Request) {
        a.writeJSON(w, r, http.StatusOK, map[string]string{"status": "ok"})
}

// writeJSON encodes body before writing any header, so a body that can't be
// encoded becomes a 500 envelope instead of an empty 200.
func (a *api) writeJSON(w http.ResponseWriter, r *http.Request, status int, body any) {
        data, err := json.Marshal(body)
        if err != nil {
                a.writeError(w, r, fmt.Errorf("encode response: %w", err))
                return
        }
        w.Header().Set("Content-Type", "application/json")
        w.WriteHeader(status)
        _, _ = w.Write(append(data, '\n'))
}

package httpapi

import (
        "fmt"
        "net/http"
        "runtime/debug"
        "strings"
        "time"
)

// recoverPanics turns a panic in next into a 500 envelope. The panic value
// and stack go to the log, never to the response.
func (a *api) recoverPanics(next http.Handler) http.Handler {
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                defer func() {
                        v := recover()
                        if v == nil {
                                return
                        }
                        if v == http.ErrAbortHandler {
                                panic(v) // the server aborts the response on purpose
                        }
                        a.writeError(w, r, fmt.Errorf("panic: %v\n%s", v, debug.Stack()))
                }()
                next.ServeHTTP(w, r)
        })
}

// serviceOrApp sends /api/... and /healthz to the service and every other
// path to the app, so the app never shadows the API.
func serviceOrApp(service, app http.Handler) http.Handler {
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                p := r.URL.Path
                if p == "/api" || strings.HasPrefix(p, "/api/") || p == "/healthz" {
                        service.ServeHTTP(w, r)
                        return
                }
                app.ServeHTTP(w, r)
        })
}

// route serves r from mux, and answers a path or method that no pattern
// matches with an error envelope instead of the mux's plain-text reply.
func (a *api) route(mux *http.ServeMux) http.Handler {
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                fallback, pattern := mux.Handler(r)
                if pattern != "" {
                        mux.ServeHTTP(w, r)
                        return
                }
                // The mux's own fallback knows whether this is a 404 or a 405 and
                // which methods are allowed; keep that and drop its body.
                miss := &missRecorder{header: http.Header{}}
                fallback.ServeHTTP(miss, r)
                if miss.status == http.StatusMethodNotAllowed {
                        allow := miss.header.Get("Allow")
                        w.Header().Set("Allow", allow)
                        a.writeError(w, r, methodNotAllowed(r, allow))
                        return
                }
                a.writeError(w, r, notFound(r))
        })
}

// missRecorder captures the status and headers of the mux's fallback reply.
type missRecorder struct {
        header http.Header
        status int
}

func (m *missRecorder) Header() http.Header         { return m.header }
func (m *missRecorder) WriteHeader(status int)      { m.status = status }
func (m *missRecorder) Write(b []byte) (int, error) { return len(b), nil }

// logRequests writes one INFO line per request with its method, path, status
// and duration.
func (a *api) logRequests(next http.Handler) http.Handler {
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                start := time.Now()
                rec := &statusRecorder{ResponseWriter: w, status: http.StatusOK}
                next.ServeHTTP(rec, r)
                a.logger.Info("request",
                        "method", r.Method,
                        "path", r.URL.Path,
                        "status", rec.status,
                        "duration", time.Since(start),
                )
        })
}

// statusRecorder remembers the status written through it. A handler that
// writes a body without a status has answered 200.
type statusRecorder struct {
        http.ResponseWriter
        status      int
        wroteHeader bool
}

func (s *statusRecorder) WriteHeader(status int) {
        if !s.wroteHeader {
                s.status, s.wroteHeader = status, true
        }
        s.ResponseWriter.WriteHeader(status)
}

func (s *statusRecorder) Write(b []byte) (int, error) {
        s.wroteHeader = true
        return s.ResponseWriter.Write(b)
}

// Unwrap lets http.ResponseController reach the server's writer.
func (s *statusRecorder) Unwrap() http.ResponseWriter { return s.ResponseWriter }

import (
        "io/fs"
        "net/http"
        "os"
        "path"
        "strings"
)

// staticHandler serves the built app from dir. Any path that isn't a file
// falls back to index.html so the app can route. NewHandler never sends it
// /api/... or /healthz.
func staticHandler(dir string) http.Handler {
        fsys := os.DirFS(dir)
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                name := strings.TrimPrefix(path.Clean(r.URL.Path), "/")
                if info, err := fs.Stat(fsys, name); err != nil || info.IsDir() {
                        name = "index.html"
                }
                http.ServeFileFS(w, r, fsys, name)
        })
}
